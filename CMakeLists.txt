cmake_minimum_required(VERSION 3.20)
project(nocturne_recomp C)

# The lifted code models a 32-bit machine but is portable C, so the host is built
# x64: the target's whole VA range (0x00400000 + 42 MB of image, plus the stack
# above it) then sits far below anything the host itself occupies, and the 1:1
# mapping in recomp_runtime.c holds with g_mem_base at 0.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure for x64; a 32-bit host collides with the mapped image.")
endif()

set(CMAKE_C_STANDARD 11)

# pcrecomp checkout: shared runtime + lifters, a sibling of this repo.
set(PCRECOMP "${CMAKE_SOURCE_DIR}/../tools" CACHE PATH "pcrecomp checkout")
if(NOT EXISTS "${PCRECOMP}/runtime/recomp32/image_loader.c")
  message(FATAL_ERROR "pcrecomp not found at ${PCRECOMP} -- clone "
                      "https://github.com/sp00nznet/pcrecomp next to this repo "
                      "or pass -DPCRECOMP=<path>.")
endif()

# Generated by run_lift.py; not in the repo (it is derived from your own binary).
file(GLOB GEN_SOURCES "${CMAKE_SOURCE_DIR}/src/recomp/gen/*.c")
if(NOT GEN_SOURCES)
  message(FATAL_ERROR "src/recomp/gen is empty -- run:\n"
                      "  py -3.11 run_lift.py analysis/nocturne.exe src/recomp/gen\n"
                      "  py -3.11 gen_imports.py")
endif()

add_executable(nocturne_recomp
  ${GEN_SOURCES}
  src/runtime/imports_gen.c
  src/runtime/recomp_runtime.c
  "${PCRECOMP}/runtime/recomp32/image_loader.c"
)

target_include_directories(nocturne_recomp PRIVATE
  src/recomp/gen                       # recomp_types.h, recomp_funcs.h
  src/runtime                          # imports.h
  "${PCRECOMP}/runtime/recomp32"       # image_loader.h
)

if(MSVC)
  # 885,000 lines of generated C: /MP is the difference between a coffee and a
  # lunch. Nothing here is performance-sensitive enough to want /O2 while the
  # bring-up is still reading stack traces.
  target_compile_options(nocturne_recomp PRIVATE /MP /W1)
endif()

target_link_libraries(nocturne_recomp PRIVATE kernel32 user32)
